#include <iostream>
#include <chrono>

using namespace std;
using namespace chrono;

// Iterative factorial function
unsigned long long iterativeFact(int value)
{
    unsigned long long fact = 1;

    while (value > 1)
    {
        fact *= value;
        value--;
    }

    return fact;
}

// Recursive factorial function
unsigned long long recursiveFact(int value)
{
    return (value <= 1) ? 1 : value * recursiveFact(value - 1);
}

int main()
{
    int number;

    cout << "Enter a non-negative integer (e.g., 20): ";

    cin >> number;

    if (cin.fail() || number < 0)
    {
        cout << "Invalid input! Please enter a non-negative integer." << endl;
        return 1;
    }

    high_resolution_clock::time_point begin1 = high_resolution_clock::now();
    unsigned long long ans1 = iterativeFact(number);
    high_resolution_clock::time_point finish1 = high_resolution_clock::now();

    auto time1 = duration_cast<nanoseconds>(finish1 - begin1);

    high_resolution_clock::time_point begin2 = high_resolution_clock::now();
    unsigned long long ans2 = recursiveFact(number);
    high_resolution_clock::time_point finish2 = high_resolution_clock::now();

    auto time2 = duration_cast<nanoseconds>(finish2 - begin2);

    cout << "\n--- Results for " << number << "! ---" << endl;

    cout << "Iterative Result : " << ans1 << endl;
    cout << "Iterative Time   : " << time1.count() << " ns" << endl;

    cout << "-------------------------------" << endl;

    cout << "Recursive Result : " << ans2 << endl;
    cout << "Recursive Time   : " << time2.count() << " ns" << endl;

    return 0;
}
