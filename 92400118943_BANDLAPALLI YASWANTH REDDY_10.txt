#include <iostream>
using namespace std;

struct Edge
{
    int u, v, weight;
};

int parent[10];

int find(int x)
{
    while (parent[x] != x)
        x = parent[x];

    return x;
}

void unionSet(int a, int b)
{
    parent[find(a)] = find(b);
}

int main()
{
    int n = 5;
    int e = 7;

    Edge edges[7] = {
        {0, 1, 2},
        {0, 2, 3},
        {1, 2, 4},
        {1, 3, 5},
        {2, 3, 1},
        {2, 4, 6},
        {3, 4, 2}
    };

    for (int i = 0; i < n; i++)
        parent[i] = i;

    for (int i = 0; i < e - 1; i++)
    {
        for (int j = 0; j < e - i - 1; j++)
        {
            if (edges[j].weight > edges[j + 1].weight)
            {
                Edge temp = edges[j];
                edges[j] = edges[j + 1];
                edges[j + 1] = temp;
            }
        }
    }

    int count = 0;
    int minCost = 0;

    cout << "Edges in Minimum Spanning Tree:\n";

    for (int i = 0; i < e; i++)
    {
        int u = edges[i].u;
        int v = edges[i].v;

        if (find(u) != find(v))
        {
            cout << u << " - " << v << " : " << edges[i].weight << endl;

            minCost += edges[i].weight;
            unionSet(u, v);
            count++;

            if (count == n - 1)
                break;
        }
    }

    cout << "Minimum Cost = " << minCost << endl;

    return 0;
}
