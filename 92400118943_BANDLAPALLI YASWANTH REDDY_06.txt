#include <iostream>
#include <climits>
using namespace std;

int main()
{
    // Dimensions of three matrices:
    // A = 10 x 20
    // B = 20 x 30
    // C = 30 x 40

    int p[] = {10, 20, 30, 40};

    // Number of matrices
    int n = 3;

    // DP table to store minimum multiplication cost
    int dp[n][n];

    // Initialize diagonal elements to 0
    for (int i = 0; i < n; i++)
    {
        dp[i][i] = 0;
    }

    // Length of matrix chain
    for (int len = 2; len <= n; len++)
    {
        // Starting matrix
        for (int i = 0; i <= n - len; i++)
        {
            // Ending matrix
            int j = i + len - 1;

            // Set initial value to maximum
            dp[i][j] = INT_MAX;

            // Find the best position to divide the matrices
            for (int k = i; k < j; k++)
            {
                int cost = dp[i][k]
                         + dp[k + 1][j]
                         + p[i] * p[k + 1] * p[j + 1];

                // Store minimum cost
                if (cost < dp[i][j])
                {
                    dp[i][j] = cost;
                }
            }
        }
    }

    // Display the minimum number of multiplications
    cout << "Minimum number of scalar multiplications = "
         << dp[0][n - 1];

    return 0;
}
