#include <iostream>
#include <vector>
#include <cstdlib>
#include <ctime>
#include <chrono>
#include <algorithm>

using namespace std;
using namespace chrono;

void maxHeapify(vector<int>& arr, int size, int root)
{
    int largest = root;
    int leftChild = 2 * root + 1;
    int rightChild = 2 * root + 2;

    if (leftChild < size && arr[leftChild] > arr[largest])
        largest = leftChild;

    if (rightChild < size && arr[rightChild] > arr[largest])
        largest = rightChild;

    if (largest != root)
    {
        swap(arr[root], arr[largest]);
        maxHeapify(arr, size, largest);
    }
}
void maxHeapSort(vector<int>& arr)
{
    int size = arr.size();

    for (int i = size / 2 - 1; i >= 0; i--)
        maxHeapify(arr, size, i);

    for (int i = size - 1; i > 0; i--)
    {
        swap(arr[0], arr[i]);
        maxHeapify(arr, i, 0);
    }
}

void minHeapify(vector<int>& arr, int size, int root)
{
    int smallest = root;
    int leftChild = 2 * root + 1;
    int rightChild = 2 * root + 2;

    if (leftChild < size && arr[leftChild] < arr[smallest])
        smallest = leftChild;

    if (rightChild < size && arr[rightChild] < arr[smallest])
        smallest = rightChild;

    if (smallest != root)
    {
        swap(arr[root], arr[smallest]);
        minHeapify(arr, size, smallest);
    }
}


void minHeapSort(vector<int>& arr)
{
    int size = arr.size();

    for (int i = size / 2 - 1; i >= 0; i--)
        minHeapify(arr, size, i);

    for (int i = size - 1; i > 0; i--)
    {
        swap(arr[0], arr[i]);
        minHeapify(arr, i, 0);
    }

    reverse(arr.begin(), arr.end());
}

int main()
{
    int n;

    cout << "Enter number of elements: ";
    cin >> n;

    srand(time(0));

    vector<int> originalArray(n);

    for (int i = 0; i < n; i++)
        originalArray[i] = rand() % 100000;

    vector<int> maxArray = originalArray;
    vector<int> minArray = originalArray;

    auto startMax = high_resolution_clock::now();
    maxHeapSort(maxArray);
    auto endMax = high_resolution_clock::now();

    auto startMin = high_resolution_clock::now();
    minHeapSort(minArray);
    auto endMin = high_resolution_clock::now();

    auto maxNano = duration_cast<nanoseconds>(endMax - startMax);
    auto maxMicro = duration_cast<microseconds>(endMax - startMax);

    auto minNano = duration_cast<nanoseconds>(endMin - startMin);
    auto minMicro = duration_cast<microseconds>(endMin - startMin);

    cout << "\n========== MAX HEAP SORT ==========\n";
    cout << "Nanoseconds  : " << maxNano.count() << " ns\n";
    cout << "Microseconds : " << maxMicro.count() << " us\n";

    cout << "\n========== MIN HEAP SORT ==========\n";
    cout << "Nanoseconds  : " << minNano.count() << " ns\n";
    cout << "Microseconds : " << minMicro.count() << " us\n";

    return 0;
}
