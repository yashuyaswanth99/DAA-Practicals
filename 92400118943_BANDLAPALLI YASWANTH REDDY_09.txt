#include <iostream>
using namespace std;

#define INF 9999

int main()
{
    int n = 5;

    int cost[5][5] = {
        {0, 2, 3, 0, 0},
        {2, 0, 4, 5, 0},
        {3, 4, 0, 1, 6},
        {0, 5, 1, 0, 2},
        {0, 0, 6, 2, 0}
    };

    int visited[5] = {0};

    visited[0] = 1;

    int edges = 0;
    int minCost = 0;

    cout << "Edges in Minimum Spanning Tree:\n";

    while (edges < n - 1)
    {
        int min = INF;
        int u = -1;
        int v = -1;

        for (int i = 0; i < n; i++)
        {
            if (visited[i])
            {
                for (int j = 0; j < n; j++)
                {
                    if (!visited[j] && cost[i][j] != 0 && cost[i][j] < min)
                    {
                        min = cost[i][j];
                        u = i;
                        v = j;
                    }
                }
            }
        }

        cout << u << " - " << v << " : " << min << endl;

        minCost += min;
        visited[v] = 1;
        edges++;
    }

    cout << "Minimum Cost = " << minCost << endl;

    return 0;
}
